require('dotenv').config(); const { Bot, InlineKeyboard } = require('grammy'); // Server faylini ulash require('./server'); // Bot ob'ektini yaratish const bot = new Bot(process.env.BOT_TOKEN); // /start komandasi kelganda bot.command('start', async (ctx) => { const userId = ctx.from.id; const firstName = ctx.from.first_name || 'Foydalanuvchi'; // Referal va ulashish havolalari const refLink = `https://t.me/TEKIN_SOVCHI_BOT?start=${userId}`; const shareText = encodeURIComponent("TEKIN SOVCHI botida o'z juftingizni toping!"); const shareUrl = `https://t.me/share/url?url=${encodeURIComponent(refLink)}&text=${shareText}`; const webAppUrl = process.env.WEB_APP_URL || "https://tekin-sovchi.onrender.com"; // Yulduzchalar o'rniga emojilar va chiroyli formatlangan matn const messageText = ` Xush kelibsiz, *${firstName}*! 💬 *Suhbatlashish tartibi:* • Yozish uchun kamida 👥 *5 ta do'stingizni* taklif qiling. • 5 ta taklif uchun 💌 *3 ta kishiga* yozish imkoni beriladi. • Keyinchalik har ➕ *1 ta taklif* uchun 💌 *3 ta* yozish imkoni beriladi. 🔗 *Sizning taklif havolangiz:* ${refLink} `; // Inline tugmalar const keyboard = new InlineKeyboard() .webApp("🚀 Mini App'ni ochish", webAppUrl) .row() .url("📲 Do'stlarga ulashish", shareUrl); // Xabarni yuborish (parse_mode: 'Markdown' bilan yulduzchalar qalin matnga aylanadi) await ctx.reply(messageText, { parse_mode: 'Markdown', reply_markup: keyboard, disable_web_page_preview: true }); }); // Botni ishga tushirish bot.start(); console.log('TEKIN SOVCHI boti muvaffaqiyatli ishga tushirildi...');
